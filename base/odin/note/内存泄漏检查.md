

#### 代码一

```go
package main

import "core:fmt"
import "core:mem"

main :: proc() {
    track: mem.Tracking_Allocator
    mem.tracking_allocator_init(&track, context.allocator)
    
    // 正确答案：最新版中这里改成了 mem.tracking_allocator(&track)
    context.allocator = mem.tracking_allocator(&track)

    defer {
        if len(track.allocation_map) > 0 {
            fmt.eprintf("=== 警告：检测到内存泄漏！ ===\n")
            for _, value in track.allocation_map {
                fmt.eprintf("- 泄漏位置: %v, 大小: %v 字节\n", value.location, value.size)
            }
        }
        if len(track.bad_free_array) > 0 {
            fmt.eprintf("=== 警告：检测到错误的内存释放！ ===\n")
            for value in track.bad_free_array {
                fmt.eprintf("- 错误释放位置: %v\n", value.location)
            }
        }
        mem.tracking_allocator_destroy(&track)
    }

    // --- 测试代码 ---
    leaked_str := fmt.aprintf("测试泄漏 22 字节")
    // defer delete(leaked_str) // 保持注释以验证泄漏检测功能
}

```

#### 代码二:显示堆栈

<font color=red>debug模式运行:</font> 

`odin run test.odin -file -debug` 

```go
package main

import "core:fmt"
import "core:debug/trace" // 最新版 Odin 使用这个包来追踪调用栈

main :: proc() {
    // 只有在调试阶段，这段内存泄漏追踪才会生效
    when ODIN_DEBUG {
        // 1. 初始化最新的追踪分配器
        track: trace.Tracking_Allocator
        trace.tracking_allocator_init(&track, context.allocator)
        context.allocator = trace.tracking_allocator(&track)

        // 2. 注册 defer，在程序退出时自动打印“泄漏在哪一行”的完整调用链
        defer {
            trace.tracking_allocator_print_results(&track)
            trace.tracking_allocator_destroy(&track)
        }
    }

    // --- 你的测试代码  ---
    fmt.println("程序开始运行...")

    // 故意制造一个泄漏：申请了内存但由于注释了 delete 导致泄漏
    leaked_str := fmt.aprintf("测试泄漏 22 字节") 
    // defer delete(leaked_str) 

    fmt.println("程序运行结束，准备退出并触发检查...")
}
```
