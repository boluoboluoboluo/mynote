#### 思路

```sh
#一阶贝塞尔曲线(实际为线段)
2个点p0,p1
线段p0p1,按比例t取点,该点的方程为:
	B(t) = p0 + (p1-p0)*t	#一阶贝塞尔方程
	B(t) = (1-t)*p0 + t*p1

#二阶贝塞尔曲线
3个点,初始点p0,控制点p1,终点p2
1.向量p0p1,即线段p0p1中间取点q0,使p0q0和q0p1的比例为t
	q0(t) = (1-t)*p0 + t*p1
2.同理,线段p1p2中间取点q1,使p1q1和q1p2的比例为t
	q1(t) = (1-t)*p1 + t*p2
3.最后,线段q0q1同样按比例t取点,该点即为曲线的点
	B2(t) = (1-t)*q0(t) + t*q1(t)
	#展开:
	B2(t) = (1-t)^2*p0 + 2*t*(1-t)*p1 + t^2*p2 	# 二阶贝塞尔方程	(0 <= t <=1)
	通过p0,p1,p2的向量坐标(x,y)带入,可计算出曲线坐标

```

```py
import matplotlib.pyplot as plt
import numpy as np

# 1. 定义 3 个控制点的坐标 (x, y)
# P0 是起点，P1 是控制点，P2 是终点
P0 = np.array([0, 0])
P1 = np.array([100, 200])
P2 = np.array([200, 0])

# 2. 让时间参数 t 从 0 匀速变到 1，分成 100 个微小的步长
t_values = np.linspace(0, 1, 100)

curve_points = []

# 3. 循环计算每一个 t 时刻对应的 (x, y) 坐标
for t in t_values:
    # 严格执行二阶贝塞尔曲线的拆解数学方程
    x_t = (1 - t)**2 * P0[0] + 2 * t * (1 - t) * P1[0] + t**2 * P2[0]
    y_t = (1 - t)**2 * P0[1] + 2 * t * (1 - t) * P1[1] + t**2 * P2[1]
    
    curve_points.append([x_t, y_t])

# 转换为 numpy 数组方便切片取值
curve_points = np.array(curve_points)

# 4. 绘图展示
plt.figure(figsize=(8, 6))

# 画出光滑的贝塞尔曲线（蓝色实线）
plt.plot(curve_points[:, 0], curve_points[:, 1], 'b-', linewidth=2, label='Bezier Curve')

# 画出控制点和控制线（红色虚线和圆点）
control_points = np.array([P0, P1, P2])
plt.plot(control_points[:, 0], control_points[:, 1], 'ro--', label='Control Polygons')

# 标注控制点名字
for i, p in enumerate([P0, P1, P2]):
    plt.text(p[0] + 5, p[1], f'P{i}', fontsize=12)

plt.title('Bézier Curve Generation in Python')
plt.legend()
plt.grid(True)
plt.show()

```
