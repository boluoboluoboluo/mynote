from functools import wraps

# 1. 定义一个通用的错误捕获装饰器
def safe_execute(default_value=None):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            try:
                return func(*args, **kwargs)
            except Exception as e:
                print(f"警告: {func.__name__} 发生错误: {e}")
                return default_value
        return wrapper
    return decorator

# 2. 业务代码里只需要优雅地“挂上”它，正文毫无杂质
@safe_execute(default_value="默认数据")
def fetch_api_data():
    # 这里故意引发一个错误，比如除以 0 或是网络错误
    return 1 / 0 

print(fetch_api_data())  # 输出: 默认数据 (程序不会崩溃)
